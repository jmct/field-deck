\begin{tikzpicture}
\begin{axis}[
  jmct axis,
  ybar interval,
  xmin=10, xmax=90, ymin=0,
  xtick={10,20,...,90},
  xlabel={Value},
  ylabel={Count},
  title={\charttitle{Placeholder: a distribution}{500 random samples from data/samples.csv}},
]
  \addplot[fill=series1, draw=canvas, line width=0.6pt,
           hist={bins=16, data min=10, data max=90}]
    table[y=value, col sep=comma]{data/samples.csv};

  % A reference line, e.g. the mean
  \draw[ink, dashed, line width=0.8pt]
    (axis cs:50,0) -- (axis cs:50,95) node[above, font=\scriptsize] {mean};
